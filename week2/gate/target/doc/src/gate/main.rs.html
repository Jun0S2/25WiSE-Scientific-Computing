9/// 2-bit add with carry
10///
11/// | a | b | ci | so | co |
12/// |---|---|----|----|----|
13/// | 0 | 0 |  0 |  0 |  0 |
14/// | 0 | 0 |  1 |  1 |  0 |
15/// | 0 | 1 |  0 |  1 |  0 |
16/// | 1 | 0 |  0 |  1 |  0 |
17/// | 0 | 1 |  1 |  0 |  1 |
18/// | 1 | 0 |  1 |  0 |  1 |
19/// | 1 | 1 |  0 |  0 |  1 |
20/// | 1 | 1 |  1 |  1 |  1 |
21/// |---|---|----|----|----|
22fn fa(a: bool, b: bool, ci: bool) -> (bool, bool) { /* Full Adder */
23    let so = a ^ b ^ ci; /* sum output */
24    let co = a && b || ci && (a ^ b); /* carry output */
25    
26    (so, co)
27}
28
29/// 2-bit multiply with carry
30///
31/// | x | y | si | ci | b | co | so |
32/// |---|---|----|----|---|----|----|
33/// | 0 | * |  0 |  0 | 0 |  0 |  0 |
34/// | * | 0 |  0 |  0 | 0 |  0 |  0 |
35/// | 0 | * |  0 |  1 | 0 |  0 |  1 |
36/// | * | 0 |  0 |  1 | 0 |  0 |  1 |
37/// | 0 | * |  1 |  0 | 0 |  0 |  1 |
38/// | * | 0 |  1 |  0 | 0 |  0 |  1 |
39/// | 0 | * |  1 |  1 | 0 |  1 |  0 |
40/// | * | 0 |  1 |  1 | 0 |  1 |  0 |
41/// | 1 | 1 |  0 |  0 | 1 |  0 |  1 |
42/// | 1 | 1 |  0 |  1 | 1 |  1 |  1 |
43/// | 1 | 1 |  1 |  * | 1 |  1 |  0 |
44/// |---|---|----|----|---|----|----|
45fn ma(x: bool, y: bool, ci: bool, si: bool) -> (bool, bool) {
46    fa(si, x && y, ci) // previous sum, x&&y(current digit), previous carry
47}

49/// add two times 4 bits with carry
50// allow : clippy
51/*
52   c3   c2   c1   c0
53    ↑    ↑    ↑    ↑
54  a3   a2   a1   a0
55+ b3 + b2 + b1 + b0
56---------------------
57  s3   s2   s1   s0
58
59*/
60#[allow(clippy::too_many_arguments)] 
61fn add4(
62    a0: bool, a1: bool, a2: bool, a3: bool,
63    b0: bool, b1: bool, b2: bool, b3: bool) 
64    -> (bool, bool, bool, bool, bool) {
65    let (s0, c0) = fa(a0, b0, false); // lsb. and there is no carry-in
66    let (s1, c1) = fa(a1, b1, c0);
67    let (s2, c2) = fa(a2, b2, c1);
68    let (s3, c3) = fa(a3, b3, c2);   // c3 : overflow bit
69
70    (s0, s1, s2, s3, c3)
71}

73/// multiply two times 4 bits giving 8 bits
74/*
75	- x0 : lsb of x
76	- x3 : msb of x
77	- y0 ~ y3 : bits of y
78
79			x3  x2  x1  x0
80		x   y3  y2  y1  y0
81		-------------------
82			p0  p0  p0  p0    (x * y0)
83		p1  p1  p1  p1   0    (x * y1, shifted left by 1)
84	p2  p2  p2  p2  0    0    (x * y2, shifted left by 2)
85*/
86
87#[allow(clippy::too_many_arguments)]
88fn mul4x4(
89    x0: bool, x1: bool, x2: bool, x3: bool,
90    y0: bool, y1: bool, y2: bool, y3: bool) 
91    -> (bool, bool, bool, bool, bool, bool, bool, bool) {
92
93	// First partial product (X * y0)
94	let (s0, s1, s2, s3, c0) = add4(x0 && y0, x1 && y0, x2 && y0, x3 && y0, false, false, false, false); // tuple  : efficient to use when there are many return valeus
95	// Second partial product (X * y1), shift one space and then add
96	let (s1b, s2b, s3b, s4, c1) = add4(s1, s2, s3, c0, x0 && y1, x1 && y1, x2 && y1, x3 && y1);
97	//  Third partial product
98	let (s2c, s3c, s4c, s5, c2) = add4(s2b, s3b, s4, c1, x0 && y2, x1 && y2, x2 && y2, x3 && y2);
99	// Fourth partial product
100	let (s3d, s4d, s5d, s6, c3) = add4(s3c, s4c, s5, c2,x0 && y3, x1 && y3, x2 && y3, x3 && y3);
101	// 마지막 carry까지 합쳐서 8비트 결과 생성
102	(s0, s1b, s2c, s3d, s4d, s5d, s6, c3)
103}

105/// Split an integer between 0 and 15 into 4 bits
106fn split4(x: u8) -> (bool, bool, bool, bool) {
107    assert!(x < 16);
108    
109    (x & 1 != 0, x & 2 != 0, x & 4 != 0, x & 8 != 0)
110}
111
112/// Merge 4 bits giving an integer
113fn merge4(x0 : bool, x1: bool, x2: bool, x3: bool) -> u8 {
114	(  if x0 { 1 } else { 0 })
115	+ (if x1 { 2 } else { 0 })
116	+ (if x2 { 4 } else { 0 })
117	+ (if x3 { 8 } else { 0 })
118}
119
120/// Merge 8 bits giving an integer
121#[allow(clippy::too_many_arguments)]
122fn merge8(x : (bool, bool, bool, bool, bool, bool, bool, bool)) -> u8 {
123	merge4(x.0, x.1, x.2, x.3) + 16 * merge4(x.4, x.5, x.6, x.7)
124}

126/// Add and multiply two numbers between 0 and 15. 
127/// Example: cargo run 4 7
128fn main() {
129	// parse and check command line arguments
130	let args: Vec<String> = env::args().collect();
131
132    if args.len() < 3 {
133		eprintln!("usage: {} number number", &args[0]); 
134		std::process::exit(1);
135    }	
136    let x = args[1].parse().unwrap_or_else(|err| panic!("argument 1 \"{}\": {err}", &args[1]));
137    let y = args[2].parse().unwrap_or_else(|err| panic!("argument 2 \"{}\": {err}", &args[2]));
138
139	if x > 15 || y > 15 {
140		panic!("Argument > 15: {x} {y}");
141	}
142	
143	//	Split the number into bits 
144	let a = split4(x);
145    let b = split4(y);
146    
147    // Add the bits and print the result
148    let z = add4(a.0, a.1, a.2, a.3, b.0, b.1, b.2, b.3);
149
150	print!("{x} + {y} = ");
151	
152    if z.4 {
153		println!("Overflow!");
154    } else {
155		let r = merge4(z.0, z.1, z.2, z.3);
156		assert_eq!(x + y, r);
157		println!("{r}"); 
158    }
159    
160    // Multiply the bits and print the result
161    let z = mul4x4(a.0, a.1, a.2, a.3, b.0, b.1, b.2, b.3);
162    let r = merge8(z);
163	assert_eq!(x * y, r);
164	println!("{x} * {y} = {r}");
165}

167/// Check addition for all allowed input values
168#[test]
169fn check_add() {
170	for x in 0 ..= 15 {
171		for y in 0 ..= 15 {
172			let a = split4(x);
173			let b = split4(y);
174			let z = add4(a.0, a.1, a.2, a.3, b.0, b.1, b.2, b.3);
175			if x + y > 15 {
176				assert_eq!(z.4, true);
177			} else {
178				assert_eq!(z.4, false);
179				let r = merge4(z.0, z.1, z.2, z.3);
180				assert_eq!(x + y, r);
181			}
182		}
183	}
184}
